```ts
// Atividade recente no Dashboard: recolhida mostra as 4 últimas; expandida, tudo das últimas
// 24 horas. O que é mais antigo só aparece no histórico completo. A escolha fica guardada no
// navegador para não voltar ao padrão a cada troca de página.

export const RECOLHIDA = 4
const JANELA_MS = 24 * 3_600_000
export const CHAVE_EXPANDIDA = 'single:atividade-expandida'

export function registrosVisiveis<T extends { created_at: string }>(
  registros: T[],
  expandida: boolean,
  agora: Date = new Date(),
): T[] {
  if (!expandida) return registros.slice(0, RECOLHIDA)
  const limite = agora.getTime() - JANELA_MS
  const recentes = registros.filter((r) => new Date(r.created_at).getTime() >= limite)
  // Um dia sem movimento não deixa o painel vazio
  return recentes.length > 0 ? recentes : registros.slice(0, RECOLHIDA)
}

/** Expandir mostraria mais do que a lista recolhida? */
export function temMaisNas24h<T extends { created_at: string }>(registros: T[], agora: Date = new Date()): boolean {
  return registrosVisiveis(registros, true, agora).length > RECOLHIDA
}

export function lerExpandida(): boolean {
  try {
    return localStorage.getItem(CHAVE_EXPANDIDA) === '1'
  } catch {
    return false
  }
}

export function guardarExpandida(valor: boolean): void {
  try {
    localStorage.setItem(CHAVE_EXPANDIDA, valor ? '1' : '0')
  } catch {
    // Sem armazenamento, a escolha vale só até sair da página
  }
}
```

```ts
/** Histórico inteiro, do mais recente ao mais antigo, em lotes que crescem com "Carregar mais". */
export function useHistoricoDeAtividade(quantidade: number, ativo = true) {
  return useQuery({
    queryKey: ['activity_log', 'historico', quantidade],
    enabled: ativo,
    placeholderData: (anterior) => anterior,
    queryFn: async () => {
      const { data, error } = await supabase
        .from('activity_log')
        .select('*')
        .order('created_at', { ascending: false })
        .limit(quantidade)
      if (error) throw error
      return (data ?? []) as ActivityLog[]
    },
  })
}
```

```tsx
import { useState } from 'react'
import { Button } from '@/components/ui/Button'
import { EstadoErro } from '@/components/ui/Estado'
import { Modal } from '@/components/ui/Modal'
import { Skeleton } from '@/components/ui/Skeleton'
import ui from '@/components/ui/ui.module.css'
import { useHistoricoDeAtividade } from '@/dados/atividade'
import type { ActivityLog, Profile } from '@/types/database'
import { Atividade } from './Atividade'
import { guardarExpandida, lerExpandida, registrosVisiveis, temMaisNas24h } from './atividadeRecolhivel'
import styles from './dashboard.module.css'

const LOTE = 50

interface AtividadeRecenteProps {
  registros: ActivityLog[]
  perfilPorId: Map<string, Profile>
}

function HistoricoCompleto({ perfilPorId, onFechar }: { perfilPorId: Map<string, Profile>; onFechar: () => void }) {
  const [quantidade, setQuantidade] = useState(LOTE)
  const historico = useHistoricoDeAtividade(quantidade)
  const registros = historico.data ?? []
  // Veio o lote inteiro: pode haver mais no banco
  const podeHaverMais = registros.length === quantidade

  return (
    <Modal aberto titulo="Histórico de atividade" onFechar={onFechar}>
      {historico.isError ? (
        <EstadoErro onTentar={() => void historico.refetch()} />
      ) : historico.isLoading ? (
        <Skeleton altura="200px" />
      ) : (
        <div className={ui.formulario}>
          <Atividade registros={registros} perfilPorId={perfilPorId} />
          {podeHaverMais && (
            <Button
              variante="secundario"
              carregando={historico.isFetching}
              onClick={() => setQuantidade((atual) => atual + LOTE)}
            >
              Carregar mais
            </Button>
          )}
        </div>
      )}
    </Modal>
  )
}

/** Atividade do Dashboard: 4 últimas, as últimas 24 horas ao expandir e o histórico inteiro num modal. */
export function AtividadeRecente({ registros, perfilPorId }: AtividadeRecenteProps) {
  const [expandida, setExpandida] = useState(lerExpandida)
  const [historico, setHistorico] = useState(false)

  function alternar() {
    setExpandida((atual) => {
      guardarExpandida(!atual)
      return !atual
    })
  }

  return (
    <>
      <Atividade registros={registrosVisiveis(registros, expandida)} perfilPorId={perfilPorId} />
      <div className={styles.atividadeRodape}>
        {(expandida || temMaisNas24h(registros)) && (
          <button type="button" className={styles.atividadeAlternar} onClick={alternar}>
            {expandida ? 'Recolher' : 'Ver mais'}
          </button>
        )}
        <button type="button" className={styles.atividadeHistorico} onClick={() => setHistorico(true)}>
          Ver histórico completo
        </button>
      </div>
      {historico && <HistoricoCompleto perfilPorId={perfilPorId} onFechar={() => setHistorico(false)} />}
    </>
  )
}
```

```ts
import { diffDias } from '@/lib/datas'
import type { Tom } from '@/lib/rotulos'
import type { Client } from '@/types/database'

// Contrato de fidelidade: começa na data de início do cliente e dura 3, 6, 12 ou 15 meses.
// O selo avisa quando faltam 30 dias ou menos (amarelo) e quando já venceu (vermelho).

export const DURACOES = [3, 6, 12, 15] as const
export const DIAS_DE_AVISO = 30

const MESES = ['jan', 'fev', 'mar', 'abr', 'mai', 'jun', 'jul', 'ago', 'set', 'out', 'nov', 'dez']
const doisDigitos = (n: number) => String(n).padStart(2, '0')

export function fimDoContrato(inicio: string, meses: number): string {
  const [ano, mes, dia] = inicio.split('-').map(Number)
  const indice = mes - 1 + meses
  const anoFim = ano + Math.floor(indice / 12)
  const mesFim = (indice % 12) + 1
  const ultimoDia = new Date(anoFim, mesFim, 0).getDate()
  return `${anoFim}-${doisDigitos(mesFim)}-${doisDigitos(Math.min(dia, ultimoDia))}`
}

export type SituacaoDoContrato = 'vigente' | 'perto' | 'vencido'

export function situacaoDoContrato(fim: string, hoje: string): SituacaoDoContrato {
  const faltam = diffDias(hoje, fim)
  if (faltam < 0) return 'vencido'
  return faltam <= DIAS_DE_AVISO ? 'perto' : 'vigente'
}

/** '2027-03-15' vira 'mar/2027'. */
export function formatarMesCurto(iso: string): string {
  const [ano, mes] = iso.split('-').map(Number)
  return `${MESES[mes - 1]}/${ano}`
}

const TOM: Record<SituacaoDoContrato, Tom> = { vigente: 'cinza', perto: 'amarelo', vencido: 'vermelho' }

export function seloDoContrato(
  cliente: Pick<Client, 'data_inicio_contrato' | 'contrato_meses'>,
  hoje: string,
): { texto: string; tom: Tom } | null {
  if (!cliente.data_inicio_contrato || !cliente.contrato_meses) return null
  const fim = fimDoContrato(cliente.data_inicio_contrato, cliente.contrato_meses)
  const situacao = situacaoDoContrato(fim, hoje)
  const mes = formatarMesCurto(fim)
  return {
    texto: situacao === 'vencido' ? `Encerrou em ${mes}` : `Encerra em ${mes}`,
    tom: TOM[situacao],
  }
}
```

```ts
export interface ProgressoDoCliente {
  cliente: Pick<Client, 'id' | 'nome'>
  pago: number
  total: number
  /** 0 a 100, arredondado */
  percentual: number
}

/** Quanto cada cliente já pagou das cobranças do mês ('AAAA-MM'), sem as canceladas. Por nome. */
export function progressoPorCliente(
  clientes: Pick<Client, 'id' | 'nome'>[],
  pagamentos: Cobranca[],
  mes: string,
): ProgressoDoCliente[] {
  return clientes
    .map((cliente) => {
      const doMes = pagamentos.filter(
        (p) => p.client_id === cliente.id && p.status !== 'cancelado' && p.mes_referencia.slice(0, 7) === mes,
      )
      const total = somar(doMes.map((p) => p.valor))
      const pago = somar(doMes.map(valorRecebido))
      return { cliente: { id: cliente.id, nome: cliente.nome }, pago, total, percentual: total > 0 ? Math.round((pago / total) * 100) : 0 }
    })
    .filter((linha) => linha.total > 0)
    .sort((a, b) => a.cliente.nome.localeCompare(b.cliente.nome))
}
```

```ts
import { textoOuNull } from '@/lib/formulario'
import type { Erros, Validacao } from '@/lib/formulario'
import { parseMoeda } from '@/lib/regras'
import type { CartaoDePagamento } from './cartoes'
import { moedaParaCampo } from './cliente'

export interface FormEdicaoDePagamento {
  valor: string
  valorPago: string
  pagoEm: string
  observacao: string
}

export interface ValoresEdicao {
  valor: number
  valorPago: number
  pagoEm: string | null
  observacao: string | null
}

const VALOR_INVALIDO = 'Informe um valor como 1.500,00.'

export function formDoCartao(cartao: Pick<CartaoDePagamento, 'valor' | 'valorPago' | 'pagoEm' | 'observacao'>): FormEdicaoDePagamento {
  return {
    valor: moedaParaCampo(cartao.valor),
    valorPago: cartao.valorPago > 0 ? moedaParaCampo(cartao.valorPago) : '',
    pagoEm: cartao.pagoEm ?? '',
    observacao: cartao.observacao ?? '',
  }
}

export function validarEdicaoDePagamento(form: FormEdicaoDePagamento): Validacao<ValoresEdicao, FormEdicaoDePagamento> {
  const erros: Erros<FormEdicaoDePagamento> = {}
  const valor = parseMoeda(form.valor)
  const valorPago = form.valorPago.trim() === '' ? 0 : parseMoeda(form.valorPago)
  if (valor == null || valor <= 0) erros.valor = VALOR_INVALIDO
  if (valorPago == null) erros.valorPago = VALOR_INVALIDO
  else if (valor != null && valorPago > valor) erros.valorPago = 'O valor pago não pode passar do total.'
  if (valorPago != null && valorPago > 0 && form.pagoEm.trim() === '') erros.pagoEm = 'Informe a data do pagamento.'
  if (Object.keys(erros).length > 0 || valor == null || valorPago == null) return { erros }
  return {
    valores: { valor, valorPago, pagoEm: valorPago > 0 ? form.pagoEm : null, observacao: textoOuNull(form.observacao) },
  }
}
```

```tsx
/**
 * O seletor nativo dispara a cada movimento enquanto a pessoa arrasta; grava só quando ela
 * para por um instante, para não mandar dezenas de gravações.
 */
function SeletorDeCor({ opcao, onEscolher }: { opcao: OpcaoDeCampo; onEscolher: (cor: string) => void }) {
  const [cor, setCor] = useState(corDe(opcao))
  const espera = useRef<ReturnType<typeof setTimeout>>()
  useEffect(() => () => clearTimeout(espera.current), [])

  return (
    <input
      type="color"
      className={styles.seletorDeCor}
      aria-label={`Cor de ${opcao.rotulo}`}
      value={cor}
      onChange={(evento) => {
        const nova = evento.target.value
        setCor(nova)
        clearTimeout(espera.current)
        espera.current = setTimeout(() => onEscolher(nova), 400)
      }}
    />
  )
}
```

```sql
create or replace function public.converter_lead(
  p_lead_id uuid, p_nome text, p_mrr numeric, p_meses integer, p_inicio date
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_lead public.leads;
  v_cliente uuid;
begin
  if not private.is_lideranca() then
    raise exception 'Só a liderança converte leads.' using errcode = '42501';
  end if;
  if p_nome is null or trim(p_nome) = '' or p_mrr is null or p_mrr < 0
     or p_meses not in (3, 6, 12, 15) or p_inicio is null then
    raise exception 'Dados da conversão incompletos.' using errcode = '22023';
  end if;

  select * into v_lead from public.leads
  where id = p_lead_id and convertido_em is null
  for update;
  if not found then
    raise exception 'Lead não encontrado ou já convertido.' using errcode = 'P0002';
  end if;

  insert into public.clients (nome, status, mrr, data_inicio_contrato, contrato_meses, contato_nome, contato_telefone)
  values (trim(p_nome), 'ativo', p_mrr, p_inicio, p_meses, v_lead.nome, v_lead.telefone)
  returning id into v_cliente;

  update public.leads set convertido_em = now(), client_id = v_cliente where id = p_lead_id;
  insert into public.lead_historico (lead_id, etapa_de, etapa_para, user_id)
  values (p_lead_id, v_lead.etapa, 'cliente', auth.uid());
  return v_cliente;
end $$;
```

```ts
import { textoOuNull } from '@/lib/formulario'
import type { Erros, Validacao } from '@/lib/formulario'
import { parseMoeda } from '@/lib/regras'
import type { EtapaDoLead, Lead } from '@/types/database'
import { DURACOES } from '@/features/clientes/contrato'

// CRM de leads. Quatro colunas fixas; "Cliente" nunca guarda card: soltar um lead nela abre a
// conversão, que cria o cliente e tira o lead do quadro.

export const ETAPAS_DO_LEAD: readonly { id: EtapaDoLead | 'cliente'; titulo: string }[] = [
  { id: 'leads', titulo: 'Leads' },
  { id: 'reuniao', titulo: 'Reunião' },
  { id: 'fechamento', titulo: 'Fechamento de contrato' },
  { id: 'cliente', titulo: 'Cliente' },
]

/** Link do WhatsApp com DDI do Brasil. Null se não houver um número que dê para discar. */
export function linkDoWhatsapp(telefone: string | null | undefined): string | null {
  const digitos = (telefone ?? '').replace(/\D/g, '')
  if (digitos.length < 10) return null
  const comPais = digitos.startsWith('55') && digitos.length >= 12 ? digitos : `55${digitos}`
  return `https://wa.me/${comPais}`
}

const MS_POR_DIA = 86_400_000

export function tempoNaEtapa(desde: string, agora: Date = new Date()): string {
  const dias = Math.floor((agora.getTime() - new Date(desde).getTime()) / MS_POR_DIA)
  if (dias < 1) return 'Na etapa desde hoje'
  return dias === 1 ? 'Na etapa há 1 dia' : `Na etapa há ${dias} dias`
}

/** Leads convertidos nos últimos 30 dias sobre leads criados nos últimos 30 dias. */
export function conversaoEm30Dias(
  leads: Pick<Lead, 'created_at' | 'convertido_em'>[],
  agora: Date = new Date(),
): { criados: number; convertidos: number; taxa: number | null } {
  const limite = agora.getTime() - 30 * MS_POR_DIA
  const noPeriodo = (iso: string | null) => iso != null && new Date(iso).getTime() >= limite
  const criados = leads.filter((l) => noPeriodo(l.created_at)).length
  const convertidos = leads.filter((l) => noPeriodo(l.convertido_em)).length
  return { criados, convertidos, taxa: criados > 0 ? convertidos / criados : null }
}

export interface FormConversao {
  nome: string
  mrr: string
  meses: string
  inicio: string
}

export interface ValoresConversao {
  nome: string
  mrr: number
  meses: number
  inicio: string
}

export function validarConversao(form: FormConversao): Validacao<ValoresConversao, FormConversao> {
  const erros: Erros<FormConversao> = {}
  const nome = form.nome.trim()
  const mrr = parseMoeda(form.mrr)
  const meses = Number(form.meses)
  if (nome === '') erros.nome = 'Informe o nome do cliente.'
  if (mrr == null || mrr < 0) erros.mrr = 'Informe um valor como 1.500,00.'
  if (!(DURACOES as readonly number[]).includes(meses)) erros.meses = 'Escolha a duração do contrato.'
  if (form.inicio.trim() === '') erros.inicio = 'Informe a data de início.'
  if (Object.keys(erros).length > 0 || mrr == null) return { erros }
  return { valores: { nome, mrr, meses, inicio: form.inicio } }
}

export interface FormLead {
  nome: string
  empresa: string
  telefone: string
  origem: string
}

export type ValoresLead = Pick<Lead, 'nome' | 'empresa' | 'telefone' | 'origem'>

export function validarLead(form: FormLead): Validacao<ValoresLead, FormLead> {
  const nome = form.nome.trim()
  if (nome === '') return { erros: { nome: 'Informe o nome do lead.' } }
  return {
    valores: {
      nome,
      empresa: textoOuNull(form.empresa),
      telefone: textoOuNull(form.telefone),
      origem: textoOuNull(form.origem),
    },
  }
}
```

```ts
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import { useAuth } from '@/features/auth/AuthContext'
import type { ValoresConversao, ValoresLead } from '@/features/crm/crm'
import { supabase } from '@/lib/supabase'
import type { EtapaDoLead, Lead } from '@/types/database'

// Leads do CRM. Só a liderança lê (RLS); para os demais a consulta nem é feita.
// Sem a tabela (migration 0004 não rodada), a consulta devolve null e a tela avisa.

const TABELA_AUSENTE = new Set(['PGRST205', '42P01'])

export function useLeads({ ativo = true }: { ativo?: boolean } = {}) {
  return useQuery({
    queryKey: ['leads'],
    enabled: ativo,
    queryFn: async () => {
      const { data, error } = await supabase.from('leads').select('*').order('posicao', { ascending: true })
      if (error) {
        if (TABELA_AUSENTE.has(error.code ?? '')) return null
        throw error
      }
      return (data ?? []) as Lead[]
    },
  })
}

/** Grava a mudança de etapa no histórico. Falhar aqui não desfaz a mudança. */
export async function registrarMudancaDeEtapa(leadId: string, de: EtapaDoLead | null, para: string, userId?: string) {
  await supabase.from('lead_historico').insert({ lead_id: leadId, etapa_de: de, etapa_para: para, user_id: userId ?? null })
}

export function useSalvarLead() {
  const queryClient = useQueryClient()
  const { perfil } = useAuth()
  return useMutation({
    mutationFn: async ({ id, valores, etapa, posicao }: { id?: string; valores: ValoresLead; etapa?: EtapaDoLead; posicao?: number }) => {
      if (id) {
        const { error } = await supabase.from('leads').update(valores).eq('id', id)
        if (error) throw error
        return id
      }
      const { data, error } = await supabase
        .from('leads')
        .insert({ ...valores, etapa: etapa ?? 'leads', posicao: posicao ?? 0, created_by: perfil?.id ?? null })
        .select()
        .single()
      if (error) throw error
      const criado = data as Lead
      await registrarMudancaDeEtapa(criado.id, null, criado.etapa, perfil?.id)
      return criado.id
    },
    onSettled: () => queryClient.invalidateQueries({ queryKey: ['leads'] }),
  })
}

export function useRemoverLead() {
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: async (id: string) => {
      const { error } = await supabase.from('leads').delete().eq('id', id)
      if (error) throw error
    },
    onSettled: () => queryClient.invalidateQueries({ queryKey: ['leads'] }),
  })
}

/** Converte pela função do banco, que cria o cliente e marca o lead numa transação só. */
export function useConverterLead() {
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: async ({ lead, valores }: { lead: Lead; valores: ValoresConversao }) => {
      const { data, error } = await supabase.rpc('converter_lead', {
        p_lead_id: lead.id,
        p_nome: valores.nome,
        p_mrr: valores.mrr,
        p_meses: valores.meses,
        p_inicio: valores.inicio,
      })
      if (error) throw error
      return data as string
    },
    onSettled: async () => {
      await queryClient.invalidateQueries({ queryKey: ['leads'] })
      await queryClient.invalidateQueries({ queryKey: ['clients'] })
    },
  })
}
```

```tsx
import { useState } from 'react'
import type { FormEvent } from 'react'
import { Button } from '@/components/ui/Button'
import { Campo } from '@/components/ui/Campo'
import { Modal } from '@/components/ui/Modal'
import { useToast } from '@/components/ui/Toast'
import ui from '@/components/ui/ui.module.css'
import { useRegistrarAtividade } from '@/dados/atividade'
import { useRemoverLead, useSalvarLead } from '@/dados/leads'
import type { Erros } from '@/lib/formulario'
import type { EtapaDoLead, Lead } from '@/types/database'
import { validarLead } from './crm'
import type { FormLead } from './crm'

interface LeadModalProps {
  /** Lead em edição; sem ele, cria um novo na etapa indicada */
  lead?: Lead
  etapa: EtapaDoLead
  posicao: number
  onFechar: () => void
}

export function LeadModal({ lead, etapa, posicao, onFechar }: LeadModalProps) {
  const [form, setForm] = useState<FormLead>({
    nome: lead?.nome ?? '',
    empresa: lead?.empresa ?? '',
    telefone: lead?.telefone ?? '',
    origem: lead?.origem ?? '',
  })
  const [erros, setErros] = useState<Erros<FormLead>>({})
  const salvar = useSalvarLead()
  const remover = useRemoverLead()
  const registrarAtividade = useRegistrarAtividade()
  const toast = useToast()

  const campo = (nome: keyof FormLead) => ({
    value: form[nome],
    erro: erros[nome],
    onChange: (evento: { target: { value: string } }) => setForm((atual) => ({ ...atual, [nome]: evento.target.value })),
  })

  async function aoEnviar(evento: FormEvent) {
    evento.preventDefault()
    const resultado = validarLead(form)
    if ('erros' in resultado) {
      setErros(resultado.erros)
      return
    }
    try {
      const id = await salvar.mutateAsync({ id: lead?.id, valores: resultado.valores, etapa, posicao })
      toast.sucesso(lead ? 'Lead atualizado.' : 'Lead criado.')
      if (!lead) {
        void registrarAtividade({ acao: 'lead_criado', descricao: `criou o lead "${resultado.valores.nome}"`, entidade: 'leads', entidadeId: id })
      }
      onFechar()
    } catch {
      toast.erro('Não foi possível salvar o lead.')
    }
  }

  async function excluir() {
    if (!lead) return
    try {
      await remover.mutateAsync(lead.id)
      toast.sucesso('Lead excluído.')
      onFechar()
    } catch {
      toast.erro('Não foi possível excluir o lead.')
    }
  }

  return (
    <Modal aberto titulo={lead ? 'Editar lead' : 'Novo lead'} onFechar={onFechar}>
      <form className={ui.formulario} onSubmit={aoEnviar} noValidate>
        <Campo rotulo="Nome" autoFocus {...campo('nome')} />
        <Campo rotulo="Empresa" {...campo('empresa')} />
        <div className={ui.duasColunas}>
          <Campo rotulo="Telefone" type="tel" placeholder="(27) 99999-0000" {...campo('telefone')} />
          <Campo rotulo="Origem" placeholder="Instagram, indicação..." {...campo('origem')} />
        </div>
        <div className={ui.acoes}>
          {lead && (
            <Button variante="fantasma" carregando={remover.isPending} onClick={() => void excluir()}>
              Excluir
            </Button>
          )}
          <Button variante="fantasma" onClick={onFechar}>
            Cancelar
          </Button>
          <Button type="submit" carregando={salvar.isPending}>
            Salvar
          </Button>
        </div>
      </form>
    </Modal>
  )
}
```

```tsx
import { useState } from 'react'
import { Quadro } from '@/components/quadro/Quadro'
import { proximaPosicao } from '@/components/quadro/colunas'
import { useMover } from '@/components/quadro/useMover'
import { EstadoErro, EstadoVazio } from '@/components/ui/Estado'
import { Icone } from '@/components/ui/Icone'
import { Skeleton } from '@/components/ui/Skeleton'
import { registrarMudancaDeEtapa, useLeads } from '@/dados/leads'
import { useAuth } from '@/features/auth/AuthContext'
import type { EtapaDoLead, Lead } from '@/types/database'
import { ConverterLeadModal } from './ConverterLeadModal'
import { LeadModal } from './LeadModal'
import { ETAPAS_DO_LEAD, linkDoWhatsapp, tempoNaEtapa } from './crm'
import styles from './crm.module.css'

type Formulario = { lead?: Lead; etapa: EtapaDoLead } | undefined

/** Pipeline de leads da liderança: Leads, Reunião, Fechamento de contrato e, ao fim, Cliente. */
export function CrmPage() {
  const leads = useLeads()
  const { perfil } = useAuth()
  const [formulario, setFormulario] = useState<Formulario>()
  const [convertendo, setConvertendo] = useState<Lead>()

  const mover = useMover<Lead>({
    tabela: 'leads',
    campo: 'etapa',
    sucesso: (_lead, destino) => `Lead movido para ${ETAPAS_DO_LEAD.find((e) => e.id === destino)?.titulo}.`,
    erro: 'Não foi possível mover o lead.',
    atividade: () => null,
    extras: () => ({ etapa_desde: new Date().toISOString() }),
  })

  if (leads.isError) return <EstadoErro onTentar={() => void leads.refetch()} />
  if (leads.isLoading) return <Skeleton altura="220px" raio="var(--radius)" />
  if (leads.data === null) {
    return (
      <EstadoVazio
        ilustracao="clientes"
        titulo="O CRM liga depois da migration 0004."
        texto="Peça a quem administra o Supabase para rodar supabase/migrations/0004_etapa8.sql."
      />
    )
  }

  const noPipeline = (leads.data ?? []).filter((lead) => lead.convertido_em == null)
  const posicaoNova = (etapa: EtapaDoLead) => proximaPosicao(noPipeline.filter((l) => l.etapa === etapa))

  function aoMover(lead: Lead, destino: string) {
    if (destino === lead.etapa) return
    // Cliente não guarda card: abre a conversão, e o lead só sai quando ela der certo
    if (destino === 'cliente') {
      setConvertendo(lead)
      return
    }
    mover(lead, destino, noPipeline)
    void registrarMudancaDeEtapa(lead.id, lead.etapa, destino, perfil?.id)
  }

  return (
    <div className={styles.pagina}>
      <Quadro
        colunas={ETAPAS_DO_LEAD}
        itens={noPipeline}
        colunaDe={(lead) => lead.etapa}
        tituloDe={(lead) => lead.nome}
        renderCard={(lead) => (
          <div className={styles.card}>
            <strong className={styles.nome}>{lead.nome}</strong>
            {lead.empresa && <span className={styles.empresa}>{lead.empresa}</span>}
            {lead.origem && <span className={styles.origem}>{lead.origem}</span>}
            <span className={styles.tempo}>{tempoNaEtapa(lead.etapa_desde)}</span>
          </div>
        )}
        renderRodape={(lead) => {
          const whatsapp = linkDoWhatsapp(lead.telefone)
          return lead.telefone ? (
            <div className={styles.rodape}>
              <span className={styles.telefone}>{lead.telefone}</span>
              {whatsapp && (
                <a
                  className={styles.whatsapp}
                  href={whatsapp}
                  target="_blank"
                  rel="noopener noreferrer"
                  aria-label={`WhatsApp de ${lead.nome}`}
                >
                  <Icone nome="externo" tamanho={14} />
                  WhatsApp
                </a>
              )}
            </div>
          ) : null
        }}
        podeCriar={(coluna) => coluna !== 'cliente'}
        onMover={aoMover}
        onCriar={(coluna) => setFormulario({ etapa: coluna as EtapaDoLead })}
        onAbrir={(lead) => setFormulario({ lead, etapa: lead.etapa })}
      />

      {formulario && (
        <LeadModal
          key={formulario.lead?.id ?? `novo-${formulario.etapa}`}
          lead={formulario.lead}
          etapa={formulario.etapa}
          posicao={posicaoNova(formulario.etapa)}
          onFechar={() => setFormulario(undefined)}
        />
      )}
      {convertendo && <ConverterLeadModal lead={convertendo} onFechar={() => setConvertendo(undefined)} />}
    </div>
  )
}
```
